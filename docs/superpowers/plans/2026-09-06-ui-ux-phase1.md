### Task 2: GSAP motion utility

**Files:**

**Interfaces:**

  - `slideDrawer(el: gsap.TweenTarget, direction: 'in' | 'out', onComplete?: () => void): void`

- [ ] **Step 1: Write the module**

```ts
// src/lib/motion.ts
import gsap from 'gsap'

export function prefersReducedMotion(): boolean {
  if (typeof window === 'undefined') return true
  return window.matchMedia('(prefers-reduced-motion: reduce)').matches
}

export function fadeInUp(target: gsap.TweenTarget, opts?: { delay?: number }) {
  if (prefersReducedMotion()) {
    gsap.set(target, { opacity: 1, y: 0 })
    return
  }
  gsap.fromTo(
    target,
    { opacity: 0, y: 12 },
    { opacity: 1, y: 0, duration: 0.4, ease: 'power2.out', delay: opts?.delay ?? 0 }
  )
}

export function staggerIn(targets: gsap.TweenTarget, opts?: { stagger?: number }) {
  if (prefersReducedMotion()) {
    gsap.set(targets, { opacity: 1, y: 0 })
    return
  }
  gsap.fromTo(
    targets,
    { opacity: 0, y: 12 },
    { opacity: 1, y: 0, duration: 0.4, ease: 'power2.out', stagger: opts?.stagger ?? 0.05 }
  )
}

export function slideDrawer(el: gsap.TweenTarget, direction: 'in' | 'out', onComplete?: () => void) {
  if (prefersReducedMotion()) {
    gsap.set(el, { xPercent: direction === 'in' ? 0 : -100 })
    onComplete?.()
    return
  }
  if (direction === 'in') {
    gsap.set(el, { xPercent: -100 })
    gsap.to(el, { xPercent: 0, duration: 0.28, ease: 'power2.out', onComplete })
  } else {
    gsap.to(el, { xPercent: -100, duration: 0.2, ease: 'power2.in', onComplete })
  }
}
```

```tsx
// src/components/Reveal.tsx
'use client'

import { useEffect, useRef } from 'react'
import { fadeInUp } from '../lib/motion'

type RevealProps = {
  children: React.ReactNode
  className?: string
  delay?: number
}

export function Reveal({ children, className, delay }: RevealProps) {
  const ref = useRef<HTMLDivElement>(null)

  useEffect(() => {
    if (ref.current) fadeInUp(ref.current, { delay })
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  return (
    <div ref={ref} className={className}>
      {children}
    </div>
  )
}
```

```tsx
'use client'

import { useEffect, useRef, useState } from 'react'
import { Menu } from 'lucide-react'
import { Sidebar } from './Sidebar'
import { usePathname } from 'next/navigation'
import { useAppState } from '../state/AppState'
import { slideDrawer } from '../lib/motion'

type DrawerState = 'closed' | 'open' | 'closing'

export function AppShell({ children }: { children: React.ReactNode }) {
  const [drawer, setDrawer] = useState<DrawerState>('closed')
  const drawerRef = useRef<HTMLDivElement>(null)
  const { toast } = useAppState()
  const pathname = usePathname()

  useEffect(() => {
    if (!drawerRef.current) return
    if (drawer === 'open') slideDrawer(drawerRef.current, 'in')
    if (drawer === 'closing') slideDrawer(drawerRef.current, 'out', () => setDrawer('closed'))
  }, [drawer])

  const closeDrawer = () => setDrawer((state) => (state === 'open' ? 'closing' : state))

  return (
    <div className="app-shell">
      <Sidebar />
      <button className="mobile-menu" onClick={() => setDrawer('open')} aria-label="Open navigation">
        <Menu size={22} />
      </button>
      {drawer !== 'closed' ? (
        <div className="mobile-nav-layer">
          <div className="mobile-nav-scrim" onClick={closeDrawer} />
          <div ref={drawerRef} className="mobile-nav-drawer">
            <Sidebar compact onClose={closeDrawer} />
          </div>
        </div>
      ) : null}
      <div className="app-content">
        {pathname !== '/app' && pathname !== '/app/settings' && (
          <p role="note" className="prototype-banner">
            Prototype workspace: example schedules and scores are not your personal progress. These tools are not yet connected to your new study context.
          </p>
        )}
        {children}
      </div>
      {toast ? (
        <div className="toast" role="status">
          {toast}
        </div>
      ) : null}
    </div>
  )
}
```

```tsx
import { createClient } from '../../../lib/supabase/server'
import Link from 'next/link'
import { StaggerList } from '../../../components/StaggerList'

export default async function WeekPage() {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return <p>Unauthorized</p>

  const { data: activeCycle } = await supabase
    .from('planning_cycles')
    .select('*')
    .eq('user_id', user.id)
    .eq('status', 'active')
    .maybeSingle()

  if (!activeCycle) {
    return (
      <div className="page-content" style={{ maxWidth: 1040 }}>
        <header className="page-header">
          <h1>Your week</h1>
          <p>No active plan was found.</p>
        </header>
        <Link href="/onboarding" className="primary-button">Plan your week</Link>
      </div>
    )
  }

  const { data: blocks } = await supabase
    .from('plan_blocks')
    .select('*')
    .eq('cycle_id', activeCycle.id)
    .order('starts_at', { ascending: true, nullsFirst: false })

  const planBlocks = blocks || []

  const { data: commitments } = await supabase
    .from('calendar_commitments')
    .select('*')
    .eq('user_id', user.id)
    .eq('status', 'active')
    .order('starts_at', { ascending: true })

  const activeCommitments = commitments || []

  const timedTasks = planBlocks.filter(b => b.placement === 'timed' && b.starts_at)
  const floatingTasks = planBlocks.filter(b => b.placement !== 'timed' || !b.starts_at)

  const calendarItems = [
    ...timedTasks.map(t => ({ ...t, _type: 'task' })),
    ...activeCommitments.map(c => ({ ...c, _type: 'commitment', title: c.label, duration_minutes: null }))
  ].sort((a, b) => new Date(a.starts_at).getTime() - new Date(b.starts_at).getTime())

  return (
    <div className="page-content" style={{ maxWidth: 1040 }}>
      <header className="page-header">
        <div>
          <h1>Your week</h1>
          <p>Class timing is fixed. Bridge, recall and verification blocks move around it according to evidence.</p>
        </div>
      </header>

      <div className="week-plan-columns">
        <section className="week-plan-column week-plan-column--calendar">
          <h2>Scheduled</h2>
          {calendarItems.length === 0 ? (
            <p>No fixed time commitments or timed study sessions this week.</p>
          ) : (
            <StaggerList className="week-plan-list">
              {calendarItems.map((item, idx) => (
                <article key={idx} className={`week-plan-item ${item._type === 'commitment' ? 'week-plan-item--commitment' : ''}`}>
                  <div className="week-plan-item-header">
                    <h3>{item.title}</h3>
                    <span className="week-plan-timestamp">
                      {new Date(item.starts_at).toLocaleString([], { weekday: 'short', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })}
                    </span>
                  </div>
                  {item._type === 'task' ? (
                    <>
                      <p>{item.detail}</p>
                      <div className="week-plan-tags">
                        <span className="week-plan-tag">{item.task_type}</span>
                        <span className="week-plan-tag">{item.duration_minutes} min</span>
                        <span className={`week-plan-tag ${item.execution === 'completed' ? 'week-plan-tag--done' : ''}`}>{item.execution.replace('_', ' ')}</span>
                      </div>
                    </>
                  ) : (
                    <div className="week-plan-tags">
                      <span className="week-plan-tag">{item.kind}</span>
                      <span className="week-plan-tag">Protected Life Commitment</span>
                    </div>
                  )}
                </article>
              ))}
            </StaggerList>
          )}
        </section>

        <section className="week-plan-column week-plan-column--floating">
          <h2>Floating Tasks</h2>
          {floatingTasks.length === 0 ? (
            <p>All tasks have scheduled times.</p>
          ) : (
            <StaggerList className="week-plan-list">
              {floatingTasks.map((block) => (
                <article key={block.id} className="week-plan-item week-plan-item--floating">
                  <h4>{block.title}</h4>
                  <p>{block.detail}</p>
                  <div className="week-plan-tags">
                    <span className="week-plan-chip">{block.task_type}</span>
                    <span className="week-plan-chip">{block.duration_minutes} min</span>
                    <span className="week-plan-chip">{block.priority}</span>
                    <span className={`week-plan-chip ${block.execution === 'completed' ? 'week-plan-tag--done' : ''}`}>{block.execution.replace('_', ' ')}</span>
                  </div>
                </article>
              ))}
            </StaggerList>
          )}
        </section>
      </div>
    </div>
  )
}
```
